Filter Operators
Operator Meaning Example
$gte Greater than or equal to {"is_featured": {"$gte": 0}}
$lte Less than or equal to {"is_featured": {"$lte": 100}}
$gt Strictly greater than {"is_featured": {"$gt": 0}}
$lt Strictly less than {"is_featured": {"$lt": 100}}
$eq Exact match (shorthand: pass the value directly) {"title": "example"}
$ne Not equal to {"title": {"$ne": "example"}}
$like SQL LIKE pattern match — use % as wildcard {"title": {"$like": "%keyword%"}}
$in Value must be in the provided list {"client_location": {"$in": ["a", "b"]}}
Example Request
Python · requests 
import requests

# Replace these with the credentials provided by your API platform
# (e.g. RapidAPI, your own deployment, etc.)
API_BASE_URL = "https://<your-api-host>"
API_KEY      = "<your-api-key>"

url = f"{API_BASE_URL}/v1/collections/_upwork_jobs/search"

headers = {
    "Content-Type":  "application/json",
    "X-Api-Key":     API_KEY,  # header name may vary by platform
}

# Structure the POST body however you need — all fields are optional.
# Use "filter" to narrow results, "order_by" to sort, "page_number" to paginate.
payload = {
    "select":      ["*"],
    "filter":      {
        "is_featured": {"$gte": 0, "$lte": 100},
        "payment_verified": {"$gte": 0},
        "title": {"$like": "%keyword%"},
    },
    "order_by":    [{"field": "posted_age", "direction": "asc"}],
    "page_number": 1,
}

resp = requests.post(url, json=payload, headers=headers)
print(resp.status_code)
print(resp.json())
